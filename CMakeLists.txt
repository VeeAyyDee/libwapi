cmake_minimum_required(VERSION 3.16)
set(WAPI_SOURCES src/constants.c src/crypto.c src/glue.c src/lifecycle.c src/packet.c)
if(COMMAND idf_component_register)
    idf_component_register(SRCS ${WAPI_SOURCES} INCLUDE_DIRS include PRIV_INCLUDE_DIRS src)
    set_target_properties(${COMPONENT_LIB} PROPERTIES OUTPUT_NAME wapi)
    target_compile_features(${COMPONENT_LIB} PRIVATE c_std_11)
else()
    project(clean_wapi C)
    add_library(wapi STATIC ${WAPI_SOURCES})
    target_include_directories(wapi PUBLIC include PRIVATE src)
    target_compile_features(wapi PRIVATE c_std_11)
    target_compile_options(wapi PRIVATE -Wall -Wextra -Werror)
    add_library(wapi_crypto SHARED src/constants.c src/crypto.c)
    target_include_directories(wapi_crypto PUBLIC include)
    enable_testing()
    add_executable(test_crypto tests/test_crypto.c)
    target_link_libraries(test_crypto PRIVATE wapi_crypto)
    add_test(NAME crypto COMMAND test_crypto)
    add_executable(test_protocol tests/test_protocol.c)
    target_include_directories(test_protocol PRIVATE src)
    target_link_libraries(test_protocol PRIVATE wapi)
    add_test(NAME protocol COMMAND test_protocol)
endif()
